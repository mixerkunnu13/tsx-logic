```tsx
Audio: "Reprenons exactement là où on s'est arrêté, parce que sécuriser son coussin et renégocier ses avantages, c'est la base défensive, mais ça ne construit pas un patrimoine."

import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene40BaseDefensive',
  durationInSeconds: 18,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene40BaseDefensive: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Building construction animation
  const buildLeft = interpolate(sp(50), [0, 1], [0, 1]);
  const buildRight = interpolate(sp(100), [0, 1], [0, 1]);

  return (
    <AbsoluteFill style={{ backgroundColor: '#DDDDDD', overflow: 'hidden', opacity: exitOp, fontFamily: 'Montserrat, sans-serif' }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill style={{
        background: 'linear-gradient(180deg, #DDDDDD 0%, #C5C5C5 100%)',
      }} />

      {/* Animated shape layers */}
      <div style={{
        position: 'absolute', width: 600, height: 600,
        backgroundColor: '#000000', opacity: 0.04,
        top: -200, right: -150, borderRadius: '20%',
        transform: `rotate(${frame * 0.1}deg)`,
      }} />
      <div style={{
        position: 'absolute', width: 450, height: 450,
        border: '3px solid #F62440', opacity: 0.06, borderRadius: '50%',
        bottom: -100, left: 100,
        transform: `rotate(${-frame * 0.15}deg)`,
      }} />

      {/* Motion texture - vertical lines */}
      <svg style={{ position: 'absolute', inset: 0, opacity: 0.05 }} width="1920" height="1080">
        {Array.from({ length: 30 }).map((_, i) => (
          <line key={i} x1={i * 64} y1={0} x2={i * 64} y2={1080} stroke="#000000" strokeWidth="1" />
        ))}
      </svg>

      {/* Floating particles */}
      {Array.from({ length: 24 }).map((_, i) => {
        const px = seededRandom(i + 1) * 1920;
        const py = (seededRandom(i + 30) * 1080 - frame * 0.4 + 1080) % 1080;
        return (
          <div key={`p${i}`} style={{
            position: 'absolute', left: px, top: py,
            width: 3, height: 3, borderRadius: '50%',
            backgroundColor: i % 4 === 0 ? '#F62440' : '#000000',
            opacity: 0.4,
          }} />
        );
      })}

      {/* Top label */}
      <div style={{
        position: 'absolute', top: 70, left: 90,
        opacity: sp(5),
      }}>
        <div style={{ display: 'flex', alignItems: 'center', gap: 18 }}>
          <div style={{ width: 60, height: 3, backgroundColor: '#F62440' }} />
          <span style={{
            fontFamily: 'Space Mono, monospace',
            fontSize: 26, color: '#F62440', fontWeight: 700, letterSpacing: 4,
          }}>NIVEAU SUPÉRIEUR // PATRIMOINE</span>
        </div>
      </div>

      {/* Main title */}
      <div style={{
        position: 'absolute', top: 130, left: 90,
        opacity: sp(15),
      }}>
        <h1 style={{
          fontFamily: 'Clash Display, sans-serif',
          fontSize: 127, color: '#000000', fontWeight: 900,
          margin: 0, lineHeight: 0.95, letterSpacing: -3,
        }}>
          DÉFENSE <span style={{ color: '#F62440' }}>≠</span> PATRIMOINE
        </h1>
      </div>

      {/* Two building blocks comparison */}
      <div style={{
        position: 'absolute', top: 340, left: 90, right: 90,
        display: 'flex', gap: 60, alignItems: 'flex-end', height: 520,
      }}>
        {/* LEFT - Base Defensive */}
        <div style={{
          flex: 1, position: 'relative',
          opacity: sp(40),
        }}>
          <div style={{
            fontFamily: 'Anton, sans-serif',
            fontSize: 47, color: '#000000', letterSpacing: 1,
            marginBottom: 20,
          }}>BASE DÉFENSIVE</div>

          {/* Stacked blocks - 3 small */}
          <div style={{ position: 'relative', height: 320 }}>
            {['LIVRET A', 'AVANTAGES', 'COUSSIN'].map((label, i) => {
              const blockOp = interpolate(buildLeft, [i * 0.25, i * 0.25 + 0.3], [0, 1], { extrapolateRight: 'clamp', extrapolateLeft: 'clamp' });
              return (
                <div key={i} style={{
                  position: 'absolute', bottom: i * 95, left: 0, right: 80,
                  height: 85, backgroundColor: '#000000',
                  border: '4px solid #F62440',
                  display: 'flex', alignItems: 'center', justifyContent: 'center',
                  fontFamily: 'Anton, sans-serif',
                  fontSize: 33, color: '#DDDDDD', letterSpacing: 3,
                  opacity: blockOp,
                  transform: `translateY(${interpolate(blockOp, [0, 1], [40, 0])}px)`,
                }}>{label}</div>
              );
            })}
          </div>
          <div style={{
            fontFamily: 'Bricolage Grotesque, sans-serif',
            fontSize: 30, color: '#000000', opacity: 0.7,
            marginTop: 20, fontWeight: 600,
          }}>Protège. Ne construit pas.</div>
        </div>

        {/* Center divider arrow */}
        <div style={{
          alignSelf: 'center', opacity: sp(85),
          transform: `translateY(${interpolate(sp(85), [0, 1], [30, 0])}px)`,
        }}>
          <svg width="100" height="120" viewBox="0 0 100 120">
            <line x1="10" y1="60" x2="80" y2="60" stroke="#F62440" strokeWidth="8" />
            <polygon points="70,40 95,60 70,80" fill="#F62440" />
            <text x="50" y="25" textAnchor="middle"
              fontFamily="Space Mono" fontSize="20" fill="#000000" fontWeight="700">NEXT</text>
          </svg>
        </div>

        {/* RIGHT - Patrimoine */}
        <div style={{
          flex: 1, position: 'relative',
          opacity: sp(90),
        }}>
          <div style={{
            fontFamily: 'Anton, sans-serif',
            fontSize: 47, color: '#F62440', letterSpacing: 1,
            marginBottom: 20,
          }}>PATRIMOINE</div>

          {/* Tower - tall block growing */}
          <div style={{ position: 'relative', height: 420 }}>
            {['PEA', 'PER', 'ASSURANCE-VIE', 'INVESTIR'].map((label, i) => {
              const blockOp = interpolate(buildRight, [i * 0.2, i * 0.2 + 0.25], [0, 1], { extrapolateRight: 'clamp', extrapolateLeft: 'clamp' });
              return (
                <div key={i} style={{
                  position: 'absolute', bottom: i * 100, left: 60, right: 0,
                  height: 92, backgroundColor: '#F62440',
                  border: '4px solid #000000',
                  display: 'flex', alignItems: 'center', justifyContent: 'center',
                  fontFamily: 'Anton, sans-serif',
                  fontSize: 33, color: '#000000', letterSpacing: 2, fontWeight: 700,
                  opacity: blockOp,
                  transform: `translateY(${interpolate(blockOp, [0, 1], [-40, 0])}px) scale(${interpolate(blockOp, [0, 1], [0.9, 1])})`,
                  boxShadow: '6px 6px 0 #000000',
                }}>{label}</div>
              );
            })}
          </div>
          <div style={{
            fontFamily: 'Bricolage Grotesque, sans-serif',
            fontSize: 30, color: '#F62440', fontWeight: 700,
            marginTop: 20,
          }}>Construit. Capitalise. Libère.</div>
        </div>
      </div>

      {/* Bottom tagline */}
      <div style={{
        position: 'absolute', bottom: 50, left: 90, right: 90,
        textAlign: 'center', opacity: sp(150),
      }}>
        <div style={{
          fontFamily: 'Clash Display, sans-serif',
          fontSize: 47, color: '#000000', fontWeight: 700, letterSpacing: -1,
        }}>
          Passons au <span style={{ color: '#F62440', fontWeight: 900 }}>niveau stratégique</span>.
        </div>
      </div>

      {/* Vignette overlay */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle, transparent 50%, rgba(0,0,0,0.22) 100%)',
        pointerEvents: 'none',
      }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45,
        fontFamily: 'Montserrat, sans-serif',
        fontSize: 22, fontWeight: 600,
        color: '#9E9E9E', opacity: 0.5,
      }}>Matures Finance</div>
    </AbsoluteFill>
  );
};

export default Scene40BaseDefensive;
```
