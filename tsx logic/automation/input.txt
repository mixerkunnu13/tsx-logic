  durationInSeconds: 28,
  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.08], { extrapolateRight: 'clamp' });

  // Calendar reveal - 12 months
  const months = Array.from({ length: 12 }, (_, i) => ({
    label: ['JAN','FEB','MAR','APR','MAY','JUN','JUL','AUG','SEP','OCT','NOV','DEC'][i],
    delay: 20 + i * 6,
  }));

  const titleAnim = sp(8);
  const subtitleAnim = sp(50);
  const stampRot = interpolate(sp(200), [0, 1], [-25, -12]);
  const stampScale = sp(200, { stiffness: 180 });

    <AbsoluteFill style={{ backgroundColor: '#8AA624', overflow: 'hidden', opacity: exitOp }}>

            height: 8,
            backgroundColor: '#000000',

            transform: `rotate(${i % 2 === 0 ? -0.5 : 0.5}deg)`,

        <defs>
          <pattern id="grid1" width="60" height="60" patternUnits="userSpaceOnUse">
            <path d="M 60 0 L 0 0 0 60" fill="none" stroke="#000000" strokeWidth="1"/>
          </pattern>
        </defs>
        <rect width="1920" height="1080" fill="url(#grid1)" />
      </svg>

      {/* Floating particles */}

  const { fps, durationInFrames } = useVideoConfig();

    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const chartAnim = sp(40);
  const beforeAnim = sp(360);
  const afterAnim = sp(430);

  // Build exponential curve
  const chartW = 1000;
  const chartH = 380;
  const chartX = 460;
  const chartY = 340;

    const t = i / steps;
