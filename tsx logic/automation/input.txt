import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene4_1',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene4_1: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();
  
  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Micro-animations for visual appeal
  const goldScale = sp(20, { damping: 10 });
  const shieldScale = sp(40);
  const textY = interpolate(sp(30), [0, 1], [50, 0]);
  const textOp = sp(30);

  // A component can be rendered by passing it to the component prop
  // You can also pass props to your component

  return (
    <AbsoluteFill style={{ backgroundColor: '#F5F5F5', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background adapted for WHITE theme */}
      <AbsoluteFill style={{
        background: `radial-gradient(circle at ${50 + Math.sin(frame*0.02)*10}% ${50 + Math.cos(frame*0.02)*10}%, #FFFFFF 0%, #E0E0E0 100%)`
      }} />

      {/* Motion texture (Grid) */}
      <AbsoluteFill style={{
        backgroundImage: 'linear-gradient(#D6D6D6 1px, transparent 1px), linear-gradient(90deg, #D6D6D6 1px, transparent 1px)',
        backgroundSize: '40px 40px',
        opacity: 0.3,
        transform: `translateY(${Math.sin(frame * 0.01) * 20}px)`
      }} />

      {/* Animated shape layers */}
      <div style={{
        position: 'absolute', top: '10%', left: '10%',
        width: 400, height: 400, borderRadius: '50%',
        border: '4px solid #FF0000', opacity: 0.05,
        transform: `scale(${1 + Math.sin(frame * 0.03) * 0.1})`
      }} />
      <div style={{
        position: 'absolute', bottom: '10%', right: '10%',
        width: 300, height: 300, backgroundColor: '#E9D700',
        opacity: 0.1, transform: `rotate(${frame * 0.5}deg)`
      }} />

      {/* Main Content */}
      <div style={{ display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center', height: '100%', padding: '0 100px', zIndex: 10 }}>
        
        {/* Eye-catching Micro Icon (Gold Shield) */}
        <div style={{ position: 'relative', marginBottom: 40, transform: `scale(${goldScale})` }}>
           <svg width="200" height="200" viewBox="0 0 24 24" fill="none" stroke="#FFC107" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round">
             <path d="M12 22s8-4 8-10V5l-8-3-8 3v7c0 6 8 10 8 10z" style={{ fill: '#FFF3E0', stroke: '#2D2D2D' }} />
             <circle cx="12" cy="11" r="4" style={{ fill: '#FFC107', transform: `scale(${shieldScale})`, transformOrigin: 'center' }} />
           </svg>
        </div>

        <h1 style={{
          fontFamily: "'Yantra One', sans-serif", fontSize: 130, color: '#2D2D2D',
          textAlign: 'center', margin: 0, lineHeight: 1.1,
          transform: `translateY(${textY}px)`, opacity: textOp
        }}>
          दुनिया का सबसे <br/><span style={{ color: '#FF0000', textTransform: 'uppercase' }}>पुराना बीमा</span>
        </h1>
        <p style={{
          fontFamily: "'Noto Sans Devanagari', sans-serif", fontSize: 50, color: '#555',
          textAlign: 'center', marginTop: 30, opacity: sp(45)
        }}>
          कागज़ी वादे टूटते हैं, सोना टिकता है।
        </p>
      </div>

      {/* Stick Figure Character */}
      <div style={{ position: 'absolute', bottom: 50, left: 100, transform: `translateY(${Math.sin(frame * 0.05) * 5}px) scale(${sp(50)})` }}>
        <svg width="150" height="250" viewBox="0 0 100 200" fill="none" stroke="#1C1C1C" strokeWidth="4" strokeLinecap="round" strokeLinejoin="round">
          <circle cx="50" cy="40" r="25" fill="#F5F5F5" />
          <path d="M35 35h10M55 35h10" stroke="#FFC107" strokeWidth="3" /> {/* Glasses */}
          <path d="M50 65v60M20 90l30-20 30 20M35 200l15-75 15 75" />
        </svg>
      </div>

      {/* Floating Particles */}
      {[...Array(8)].map((_, i) => (
        <div key={i} style={{
          position: 'absolute',
          left: `${seededRandom(i) * 100}%`,
