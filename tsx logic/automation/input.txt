import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene31_BuffettEntry',
  durationInSeconds: 14,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene31_BuffettEntry: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Timeline drift camera
  const camX = interpolate(frame, [0, durationInFrames], [0, -40], { extrapolateRight: 'clamp' });
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.05], { extrapolateRight: 'clamp' });

  // Stock line graph progress
  const lineProg = interpolate(frame, [40, 150], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.cubic) });

  // Graph points: rise then crater
  const pts = [
    { x: 200, y: 600 }, { x: 420, y: 540 }, { x: 640, y: 480 },
    { x: 860, y: 400 }, { x: 1080, y: 340 }, { x: 1300, y: 520 }, { x: 1520, y: 720 },
  ];
  const totalLen = pts.length - 1;
  const visibleCount = lineProg * totalLen;
  let pathD = `M ${pts[0].x} ${pts[0].y}`;
  for (let i = 1; i < pts.length; i++) {
    const seg = Math.min(1, Math.max(0, visibleCount - (i - 1)));
    if (seg <= 0) break;
    const px = pts[i - 1].x + (pts[i].x - pts[i - 1].x) * seg;
    const py = pts[i - 1].y + (pts[i].y - pts[i - 1].y) * seg;
    pathD += ` L ${px} ${py}`;
  }

  return (
    <AbsoluteFill style={{ backgroundColor: '#A1BC98', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 20%, rgba(255,87,34,0.10), transparent 55%), linear-gradient(135deg, #A1BC98 0%, #8FAC86 100%)' }} />

      {/* Animated shape layers */}
      <AbsoluteFill style={{ transform: `translateX(${camX}px) scale(${camZoom})` }}>
        <div style={{ position: 'absolute', left: 1300, top: 120, width: 480, height: 480, borderRadius: '50%', border: '2px solid rgba(238,238,238,0.4)', opacity: 0.06, transform: `rotate(${frame * 0.3}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})` }} />
        <div style={{ position: 'absolute', left: -120, top: 500, width: 420, height: 420, background: 'rgba(255,87,34,1)', opacity: 0.05, transform: `rotate(${-frame * 0.2}deg)` }} />
      </AbsoluteFill>

      {/* Motion texture — vertical timeline grid */}
      <AbsoluteFill style={{ opacity: 0.12, transform: `translateX(${camX}px)` }}>
        {Array.from({ length: 14 }).map((_, i) => (
          <div key={i} style={{ position: 'absolute', left: `${i * 7.5}%`, top: 0, width: 1, height: '100%', background: '#EEEEEE' }} />
        ))}
      </AbsoluteFill>

      {/* Floating particles */}
      {Array.from({ length: 16 }).map((_, i) => {
        const r = seededRandom(i + 1);
        const r2 = seededRandom(i + 50);
        const py = (r2 * 1080 + frame * (0.4 + r)) % 1080;
        return (
          <div key={i} style={{ position: 'absolute', left: `${r * 100}%`, top: py, width: 5, height: 5, borderRadius: '50%', background: '#FF5722', opacity: 0.35 }} />
        );
      })}

      {/* Main content with spring animations */}
      <AbsoluteFill style={{ transform: `translateX(${camX}px) scale(${camZoom})` }}>
        {/* Headline */}
        <div style={{ position: 'absolute', top: 90, left: 110, transform: `translateY(${interpolate(sp(10), [0, 1], [-60, 0])}px)`, opacity: sp(10) }}>
          <div style={{ fontFamily: 'Mont Heavy, Inter', fontSize: 102, fontWeight: 900, color: '#FF5722', lineHeight: 1, letterSpacing: -2 }}>BUFFETT'S</div>
          <div style={{ fontFamily: 'Mont Heavy, Inter', fontSize: 102, fontWeight: 900, color: '#EEEEEE', lineHeight: 1, letterSpacing: -2 }}>OIL BET</div>
        </div>

        {/* Year markers */}
        <div style={{ position: 'absolute', top: 270, left: 200, transform: `scale(${sp(50)})`, opacity: sp(50) }}>
          <div style={{ fontFamily: 'JetBrains Mono, monospace', fontSize: 58, fontWeight: 700, color: '#EEEEEE' }}>2005</div>
          <div style={{ fontFamily: 'Inter', fontSize: 30, color: '#FF5722', fontWeight: 600 }}>ENTERED</div>
        </div>
        <div style={{ position: 'absolute', top: 230, left: 1000, transform: `scale(${sp(75)})`, opacity: sp(75) }}>
          <div style={{ fontFamily: 'JetBrains Mono, monospace', fontSize: 72, fontWeight: 700, color: '#FF5722' }}>2008</div>
          <div style={{ fontFamily: 'Inter', fontSize: 32, color: '#EEEEEE', fontWeight: 600 }}>EXPANDED BIG</div>
        </div>

        {/* Stock line graph */}
        <svg width="1920" height="1080" style={{ position: 'absolute', top: 0, left: 0 }}>
          <path d={pathD} fill="none" stroke="#FF5722" strokeWidth={7} strokeLinecap="round" strokeLinejoin="round" />
          {lineProg >= 1 && (
            <circle cx={pts[4].x} cy={pts[4].y} r={14} fill="#EEEEEE" stroke="#FF5722" strokeWidth={4} opacity={interpolate(frame, [150, 165], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' })} />
          )}
        </svg>

        {/* Crater label */}
        <div style={{ position: 'absolute', top: 740, left: 1280, opacity: interpolate(frame, [160, 180], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }), fontFamily: 'Nexa Bold, Inter', fontSize: 44, fontWeight: 800, color: '#FF5722' }}>
          CRATERED ↓
        </div>
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill style={{ boxShadow: 'inset 0 0 350px rgba(28,28,28,0.55)', pointerEvents: 'none' }} />

      {/* Watermark */}
      <div style={{ position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter', fontSize: 22, fontWeight: 600, color: '#EEEEEE', opacity: 0.3 }}>Failed Finances</div>
    </AbsoluteFill>
  );
};

export default Scene31_BuffettEntry;
