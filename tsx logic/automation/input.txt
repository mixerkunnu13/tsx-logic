  // Drain bar width 100% -> 48%

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.cubic),
  });

  // Date flip 1 -> 3
  const dateNum = frame < 150 ? '01' : '03';
  const dateFlip = frame >= 150 ? sp(150) : 1;

  return (
    <AbsoluteFill style={{ backgroundColor: '#DC5F00', overflow: 'hidden' }}>
