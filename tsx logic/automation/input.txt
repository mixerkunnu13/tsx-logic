import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene31',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene31: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const camZoom = interpolate(frame, [0, durationInFrames], [1.05, 1.12], { extrapolateRight: 'clamp' });
  const titleIn = sp(10);
  const drainLevel = interpolate(frame, [40, 220], [1, 0.25], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.ease) });
  const coinFall = (i: number) => interpolate((frame - 50 - i * 12) % 90, [0, 90], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#14FFEC', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient layer */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 50% 35%, rgba(255,255,255,0.45), transparent 60%)' }} />

      {/* Shape layers — slow drift */}
      <div style={{ position: 'absolute', top: 120, left: 140, width: 520, height: 520, borderRadius: '50%', border: '3px solid #000000', opacity: 0.05, transform: `rotate(${frame * 0.2}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})` }} />
      <div style={{ position: 'absolute', bottom: 80, right: 160, width: 460, height: 460, border: '3px solid #212121', opacity: 0.06, transform: `rotate(${-frame * 0.15}deg)` }} />

      {/* Motion texture — vertical grid lines */}
      {Array.from({ length: 14 }).map((_, i) => (
        <div key={i} style={{ position: 'absolute', top: 0, left: `${i * 7.2}%`, width: 1, height: '100%', backgroundColor: '#000000', opacity: 0.04 }} />
      ))}

      {/* Floating particles */}
      {Array.from({ length: 18 }).map((_, i) => {
        const r = seededRandom(i + 1);
        const y = (frame * (0.4 + r) + i * 90) % 1080;
        return <div key={i} style={{ position: 'absolute', left: `${seededRandom(i + 5) * 100}%`, top: y, width: 6 + r * 6, height: 6 + r * 6, borderRadius: '50%', backgroundColor: '#212121', opacity: 0.12 }} />;
      })}

      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        {/* Content */}
        <div style={{ position: 'absolute', top: 90, width: '100%', textAlign: 'center', opacity: titleIn, transform: `translateY(${(1 - titleIn) * 40}px)` }}>
          <div style={{ fontFamily: 'Inter, Poppins', fontSize: 119, fontWeight: 800, color: '#000000', letterSpacing: -2 }}>CLOSING THE LEAK</div>
          <div style={{ fontFamily: 'Inter, Poppins', fontSize: 51, fontWeight: 600, color: '#212121', marginTop: 8 }}>Costing the Trust Fund billions</div>
        </div>

        {/* Draining vault graphic */}
        <div style={{ position: 'absolute', top: 360, left: '50%', transform: 'translateX(-50%)' }}>
          <svg width="420" height="480" viewBox="0 0 420 480">
            <rect x="40" y="60" width="340" height="360" rx="20" fill="none" stroke="#000000" strokeWidth="8" />
            <clipPath id="vaultClip31"><rect x="48" y="68" width="324" height="344" rx="14" /></clipPath>
            <g clipPath="url(#vaultClip31)">
              <rect x="48" y={68 + (1 - drainLevel) * 344} width="324" height={drainLevel * 344} fill="#212121" opacity="0.85" />
              {Array.from({ length: 6 }).map((_, i) => (
                <line key={i} x1="48" y1={80 + (1 - drainLevel) * 344 + i * 12} x2="372" y2={80 + (1 - drainLevel) * 344 + i * 12} stroke="#14FFEC" strokeWidth="2" opacity="0.3" />
              ))}
            </g>
            <circle cx="210" cy="240" r="34" fill="none" stroke="#000000" strokeWidth="6" />
            <line x1="210" y1="240" x2="210" y2="210" stroke="#000000" strokeWidth="6" transform={`rotate(${frame * 4} 210 240)`} />
            <rect x="180" y="420" width="60" height="30" fill="#000000" />
          </svg>

  );

          </div>
          {/* Spouse bar */}

            <div style={{ fontFamily: 'Inter, Poppins', fontSize: 64, fontWeight: 900, color: '#000000' }}>${spouseAmt.toLocaleString()}</div>

        {/* Higher earner badge */}
        <div style={{ position: 'absolute', top: 300, right: 240, opacity: badgeIn, transform: `scale(${badgeIn}) rotate(8deg)` }}>

            <div style={{ fontFamily: 'Inter, Poppins', fontSize: 51, fontWeight: 900, color: '#14FFEC' }}>+$1,000 MORE</div>

      {/* Watermark */}

    </AbsoluteFill>
  );
