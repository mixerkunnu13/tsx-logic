  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);

  id: 'Scene32MarcheArriere',
  durationInSeconds: 21,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene32MarcheArriere: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Pan camera
  const camX = interpolate(frame, [0, durationInFrames], [-30, 30], { extrapolateRight: 'clamp' });

  // Screen shake on "marche arrière" impact (~frame 180)
  const shake = frame > 175 && frame < 195 ? Math.sin(frame * 2.2) * 8 * (1 - (frame - 175) / 20) : 0;

  // Descending line path that suddenly turns up
  const lineProgress = interpolate(frame, [30, 150], [0, 1], { extrapolateRight: 'clamp', easing: Easing.inOut(Easing.cubic) });
  const reverseProgress = interpolate(frame, [165, 240], [0, 1], { extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });

  return (
    <AbsoluteFill style={{ backgroundColor: '#215E61', overflow: 'hidden', opacity: exitOp }}>
      <AbsoluteFill style={{ background: 'linear-gradient(135deg, #163F41 0%, #215E61 50%, #2C7A7E 100%)' }} />

      {/* Animated shapes */}

        );

      {[...Array(10)].map((_, i) => (

          position: 'absolute',
          left: `${i * 11 - 10}%`, top: '-10%',
          width: 1, height: '120%',
          background: '#FBF5A7', opacity: 0.04,
          transform: `rotate(18deg) translateX(${Math.sin(frame * 0.02 + i) * 14}px)`,
        }} />
      ))}

      {/* Particles */}
      {[...Array(16)].map((_, i) => {

        return (

            position: 'absolute', left: px, top: `${s * 100}%`,

        );
