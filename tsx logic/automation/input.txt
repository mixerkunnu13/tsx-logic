import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene31_GoldRise',
  durationInSeconds: 24,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene31_GoldRise: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const bgShift = interpolate(frame, [0, durationInFrames], [0, 30], { extrapolateRight: 'clamp' });
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.06], { extrapolateRight: 'clamp' });

  // Line graph rise progress
  const graphProgress = interpolate(frame, [40, 150], [0, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });

  // Counter from 3335 to 4732
  const priceVal = Math.round(interpolate(frame, [50, 160], [3335, 4732], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  }));

  const pctVal = Math.round(interpolate(frame, [120, 190], [0, 41], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  }));

  // Graph path: rising line
  const points = [
    { x: 200, y: 760 }, { x: 460, y: 690 }, { x: 720, y: 640 },
    { x: 980, y: 520 }, { x: 1240, y: 410 }, { x: 1500, y: 300 }, { x: 1700, y: 230 },
  ];
  const visibleCount = Math.max(1, Math.floor(graphProgress * (points.length - 1)) + 1);
  const partial = points.slice(0, visibleCount);
  let pathD = `M ${points[0].x} ${points[0].y}`;
  partial.forEach((p) => { pathD += ` L ${p.x} ${p.y}`; });

  return (
    <AbsoluteFill style={{ backgroundColor: '#FF5722', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background */}
      <AbsoluteFill style={{
        background: `radial-gradient(circle at ${30 + bgShift}% 40%, rgba(48,56,65,0.55) 0%, rgba(255,87,34,0) 60%)`,
      }} />

      {/* Shape Layers — drifting geometric forms */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        <div style={{
          position: 'absolute', top: 120, left: 100, width: 520, height: 520,
          border: '3px solid #303841', borderRadius: '50%', opacity: 0.05,
          transform: `rotate(${frame * 0.2}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
        }} />
        <div style={{
          position: 'absolute', bottom: 80, right: 140, width: 380, height: 380,
          backgroundColor: '#303841', opacity: 0.04,
          transform: `rotate(${-frame * 0.15}deg)`,
        }} />
      </AbsoluteFill>

      {/* Motion texture — vertical grid lines */}
      {Array.from({ length: 14 }).map((_, i) => (
        <div key={`grid-${i}`} style={{
          position: 'absolute', top: 0, bottom: 0, left: `${i * 7.2}%`, width: 1,
          backgroundColor: '#303841', opacity: 0.06,
          transform: `translateY(${Math.sin(frame * 0.02 + i) * 10}px)`,
        }} />
      ))}

      {/* Floating particles */}
      {Array.from({ length: 18 }).map((_, i) => {
        const r = seededRandom(i + 1);
        const r2 = seededRandom(i + 50);
        const px = r * 1920;
        const py = (r2 * 1080 + frame * (0.4 + r)) % 1080;
        return (
          <div key={`p-${i}`} style={{
            position: 'absolute', left: px, top: py, width: 4 + r * 5, height: 4 + r * 5,
            backgroundColor: '#F5F5F5', borderRadius: '50%', opacity: 0.18 + r * 0.15,
          }} />
        );
      })}

      {/* Main content */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        {/* Header label */}
        <div style={{
          position: 'absolute', top: 90, left: 110,
          fontFamily: 'Mont Heavy, Inter, sans-serif', fontWeight: 900,
          fontSize: 102, color: '#F5F5F5', lineHeight: 1.0,
          opacity: sp(10), transform: `translateY(${(1 - sp(10)) * -40}px)`,
        }}>
          GOLD'S CLIMB
        </div>
        <div style={{
          position: 'absolute', top: 210, left: 114,
          fontFamily: 'Inter, sans-serif', fontWeight: 600,
          fontSize: 46, color: '#303841',
          opacity: sp(25),
        }}>
          May 2025 → May 2026
        </div>

        {/* SVG line graph */}
        <svg width="1920" height="1080" style={{ position: 'absolute', top: 0, left: 0 }}>
          {/* baseline axis */}
          <line x1="180" y1="800" x2="1740" y2="800" stroke="#303841" strokeWidth="3" opacity="0.4" />
          <line x1="180" y1="200" x2="180" y2="800" stroke="#303841" strokeWidth="3" opacity="0.4" />
          {/* rising path */}
          <path d={pathD} fill="none" stroke="#F5F5F5" strokeWidth="9"
            strokeLinecap="round" strokeLinejoin="round"
            style={{ filter: 'drop-shadow(0 0 14px rgba(245,245,245,0.5))' }} />
          {/* endpoint pulse */}
          {graphProgress > 0.9 && (
            <circle cx={points[points.length - 1].x} cy={points[points.length - 1].y}
              r={14 + Math.sin(frame * 0.2) * 4} fill="#F5F5F5" />
          )}
        </svg>

        {/* Price counter */}
        <div style={{
          position: 'absolute', bottom: 150, left: 120,
          fontFamily: 'Nexa Bold, Mont Heavy, sans-serif', fontWeight: 900,
          fontSize: 187, color: '#F5F5F5', lineHeight: 0.95,
          opacity: sp(50),
        }}>
          ${priceVal.toLocaleString()}
        </div>

        {/* +41% badge */}
        <div style={{
          position: 'absolute', bottom: 220, right: 150,
          fontFamily: 'Mont Heavy, sans-serif', fontWeight: 900,
          fontSize: 119, color: '#303841',
          backgroundColor: '#F5F5F5', padding: '20px 44px', borderRadius: 18,
          opacity: sp(120), transform: `scale(${0.7 + sp(120) * 0.3})`,
          boxShadow: '0 12px 40px rgba(48,56,65,0.4)',
        }}>
          +{pctVal}%

      {/* Vignette overlay */}

};
