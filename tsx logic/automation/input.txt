  id: 'UtilitiesIntro',
  durationInSeconds: 12,
  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const { fps, durationInFrames } = useVideoConfig();

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',

  const titleY = interpolate(sp(35), [0, 1], [80, 0]);
  const lineWidth = interpolate(sp(55), [0, 1], [0, 900]);

    <AbsoluteFill style={{ backgroundColor: '#0D0B61', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient base */}

        background: 'radial-gradient(circle at 30% 40%, rgba(180,229,13,0.15) 0%, rgba(13,11,97,0) 60%)',
