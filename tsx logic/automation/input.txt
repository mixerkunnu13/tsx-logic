  id: 'Scene01_FlatlinePetrol',
  durationInSeconds: 16,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const { fps, durationInFrames } = useVideoConfig();

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',

  // Chart line draw progress (flatline animation)
  const lineProgress = interpolate(frame, [40, 180], [0, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.cubic),
  });

  // Year markers reveal
  const yearReveal = (i: number) => sp(60 + i * 18);

  // Main headline
  const headlineOp = sp(20);
  const headlineY = interpolate(sp(20), [0, 1], [60, 0]);

  // Sub headline (4 साल)
  const subOp = sp(80);
  const subScale = interpolate(sp(80), [0, 1], [0.7, 1]);

  // Bottom tagline pulse
  const taglineOp = sp(220);
  const taglinePulse = 1 + Math.sin(frame * 0.08) * 0.02;

  // Camera subtle zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.04]);

  // Chart geometry
  const chartX = 280;
  const chartY = 620;
  const chartW = 1360;
  const chartH = 80;
  const flatY = chartY;

  return (
    <AbsoluteFill style={{ backgroundColor: '#C7E8CA', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(circle at 30% 30%, rgba(250,185,91,0.18) 0%, transparent 55%), radial-gradient(circle at 75% 75%, rgba(232,237,242,0.35) 0%, transparent 50%)',
        }}
      />

      {/* Animated transparent shape layers */}
      <div
        style={{
          position: 'absolute',
          top: -200,
          right: -150,
          width: 700,
          height: 700,
          borderRadius: '50%',
          background: '#FAB95B',
          opacity: 0.06,
          transform: `scale(${1 + Math.sin(frame * 0.02) * 0.05}) rotate(${frame * 0.1}deg)`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          bottom: -250,
          left: -200,
          width: 800,
          height: 800,
          borderRadius: 80,
          background: '#E8EDF2',
          opacity: 0.25,
          transform: `rotate(${frame * 0.05}deg) scale(${1 + Math.cos(frame * 0.025) * 0.04})`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          top: '40%',
          left: '50%',
          width: 500,
          height: 500,
          border: '3px solid #1C1C1C',
          opacity: 0.05,
          transform: `translate(-50%, -50%) rotate(${frame * -0.15}deg)`,
        }}
      />

      {/* Grid texture overlay */}
      <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.08 }}>
        <defs>
          <pattern id="gridPat" width="80" height="80" patternUnits="userSpaceOnUse">
            <path d="M 80 0 L 0 0 0 80" fill="none" stroke="#1C1C1C" strokeWidth="1" />
          </pattern>
        </defs>
        <rect width="1920" height="1080" fill="url(#gridPat)" />
      </svg>

      {/* Floating particles */}
      {Array.from({ length: 22 }).map((_, i) => {

        const size = 4 + seed * 6;
        return (

            key={`p-${i}`}
            style={{
              position: 'absolute',
              left: x,
              top: y,
              width: size,
              height: size,
              borderRadius: '50%',
              background: i % 2 === 0 ? '#FAB95B' : '#1C1C1C',
              opacity: 0.25,
            }}
          />
        );
      })}

      {/* Main camera-scaled content */}
      <AbsoluteFill style={{ transform: `scale(${camScale})` }}>
        {/* TOP TAG */}
        <div
          style={{
            position: 'absolute',
            top: 90,
            left: '50%',
            transform: `translateX(-50%) translateY(${interpolate(sp(10), [0,1], [-40, 0])}px)`,

            background: '#1C1C1C',

          >
            दाम एक ही जगह

        {/* FLATLINE CHART */}
        <svg width="1920" height="1080" style={{ position: 'absolute', top: 0, left: 0 }}>
          {/* Chart base axis */}
          <line
            x1={chartX}
