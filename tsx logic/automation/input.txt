        {Array.from({ length: 14 }).map((_, i) => (
          <div key={`p-${i}`} style={{
            position: 'absolute',
            left: seededRandom(i + 211) * 1920,
            top: (seededRandom(i + 221) * 1080 - frame * 0.4 + 1080) % 1080,
            width: 3 + seededRandom(i + 18) * 5, height: 3 + seededRandom(i + 18) * 5,
            borderRadius: '50%', backgroundColor: C.PALE, opacity: 0.22,
          }} />
        ))}

        {/* Beat 1 — SHAME fades / SCORE stands */}
        <div style={{ position: 'absolute', top: 150, left: 0, right: 0, textAlign: 'center', opacity: 1 - reveal }}>
          <div style={{
            fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 160, fontWeight: 900, lineHeight: 1.04,
            color: C.DEEP, opacity: sp(8) * interpolate(frame, [90, 140], [1, 0.18], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }),
            transform: `translateY(${(1 - sp(8, { damping: 9 })) * 80}px)`,
            letterSpacing: 6,
          }}>SHAME FADES.</div>
          <div style={{
            fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 160, fontWeight: 900, lineHeight: 1.04,
            color: C.LIGHT, opacity: sp(50),
            transform: `scale(${0.8 + sp(50, { damping: 8 }) * 0.2})`,
            letterSpacing: 6, textShadow: '0 24px 60px rgba(63,79,68,0.35)',
          }}>SCORES REBUILD.</div>
          <div style={{ marginTop: 30, fontFamily: 'Inter, Poppins, sans-serif', fontSize: 40, fontWeight: 600, color: C.DEEP, opacity: sp(95) }}>
            Shame is permanent only if you let it be.
          </div>
        </div>

        {/* Beat 2 — machinery revealed behind curtains */}
        <div style={{ position: 'absolute', inset: 0, opacity: reveal }}>
          <Gear size={300} x={680} y={280} dir={1} />
          <Gear size={190} x={930} y={470} dir={-1} />
          <Gear size={140} x={620} y={540} dir={-1} />
          <div style={{
            position: 'absolute', top: 190, left: 0, right: 0, textAlign: 'center',
            fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 100, fontWeight: 900, color: C.DEEP,
            letterSpacing: 8, transform: `translateY(${(1 - reveal) * -70}px)`,
          }}>THE MACHINERY</div>

          {/* Old story collapsing */}
          <div style={{
            position: 'absolute', bottom: 170, left: 0, right: 0, textAlign: 'center',
            transform: `translateY(${collapse * 220}px) rotate(${collapse * 7}deg)`,
            opacity: 1 - collapse * 0.9,
          }}>
            <span style={{
              fontFamily: 'Inter, Poppins, sans-serif', fontSize: 46, fontWeight: 700,
              color: C.LIGHT, backgroundColor: C.TEAL, padding: '16px 44px', borderRadius: 14,
              textDecoration: collapse > 0.1 ? 'line-through' : 'none',
            }}>
              &ldquo;Borrowers just stopped paying&rdquo;
            </span>
          </div>
          <div style={{
            position: 'absolute', bottom: 110, left: 0, right: 0, textAlign: 'center',
            fontFamily: 'JetBrains Mono, monospace', fontSize: 34, fontWeight: 700, color: C.DEEP,
            letterSpacing: 4, opacity: sp(330),
          }}>
            COLLAPSES AGAINST VERIFIED EVIDENCE
          </div>
        </div>

        {/* Curtain panels — slide apart */}
        <div style={{
          position: 'absolute', top: 0, bottom: 0, left: 0, width: 970,
          backgroundColor: C.DEEP, opacity: reveal > 0.98 ? 0 : 0.94,
          transform: `translateX(${curtainL}px)`,
          borderRight: `10px solid ${C.SOFT}`,
          display: frame < 160 ? 'none' : 'block',
        }} />
        <div style={{
          position: 'absolute', top: 0, bottom: 0, right: 0, width: 970,
          backgroundColor: C.DEEP, opacity: reveal > 0.98 ? 0 : 0.94,
          transform: `translateX(${curtainR}px)`,
          borderLeft: `10px solid ${C.SOFT}`,
          display: frame < 160 ? 'none' : 'block',
        }} />

        {/* Vignette overlay */}
        <AbsoluteFill style={{
          background: 'radial-gradient(ellipse at center, rgba(0,0,0,0) 58%, rgba(28,28,28,0.28) 100%)',
          pointerEvents: 'none',
        }} />

        {/* Watermark */}
        <div style={{
          position: 'absolute', bottom: 30, right: 45,
          fontFamily: 'Inter, sans-serif', fontSize: 22, fontWeight: 600,
          color: '#EEEEEE', opacity: 0.3,
        }}>KuxAI.7x</div>
      </AbsoluteFill>
    </AbsoluteFill>
  );
};
